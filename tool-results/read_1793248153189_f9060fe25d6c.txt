    62→def top_split(s, sep):
    63→    """Split on a single-char separator at paren/brace/quote depth 0."""
    64→    parts, depth, start, i, instr = [], 0, 0, 0, False
    65→    while i < len(s):
    66→        c = s[i]
    67→        if instr:
    68→            if c == '"':
    69→                if i + 1 < len(s) and s[i + 1] == '"':
    70→                    i += 2
    71→                    continue
    72→                instr = False
    73→            i += 1
    74→            continue
    75→        if c == '"':
    76→            instr = True
    77→        elif c in "([{":
    78→            depth += 1
    79→        elif c in ")]}":
    80→            depth -= 1
    81→        elif c == sep and depth == 0:
    82→            parts.append(s[start:i])
    83→            start = i + 1
    84→        i += 1
    85→    parts.append(s[start:])
    86→    return parts

    89→def conv_type(s, ctx, nested=False):
    90→    """Convert a Luau type string to a TypeScript type string."""
    91→    s = (s or "").strip()
    92→    s = re.sub(r"\s*\|\s*\[\]\s*$", "", s).strip()   # crawler artifact
    93→    # flattened Luau line comments: drop "-- text" up to the next "field:".
    94→    s = re.sub(r"--\s+[\w\s]*?(?=\s*[a-z_]+\s*:)", "", s)
    95→    if not s or s == "?":
    96→        return "unknown"
    97→    if "|" in s and len(top_split(s, "|")) > 1:
    98→        alts, seen = [], set()
    99→        for p in top_split(s, "|"):
   100→            t = conv_type(p, ctx, True)
   101→            if " => " in t:            # TS1385: parenthesize fn types in unions
   102→                t = f"({t})"
   103→            if t and t not in seen:
   104→                seen.add(t)
   105→                alts.append(t)
   106→        return alts[0] if len(alts) == 1 else " | ".join(alts)
   107→    if "&" in s and len(top_split(s, "&")) > 1:
   108→        return " & ".join(conv_type(p, ctx, True) for p in top_split(s, "&"))
   109→    if s.endswith("?"):
   110→        base = conv_type(s[:-1], ctx, True)
   111→        return base if base in ("unknown", "any") else f"{base} | undefined"
   112→    s = s.strip()
   113→    if s.startswith("(") and s.endswith(")"):
   114→        inner = s[1:-1].strip()
   115→        if not inner:
   116→            return "void" if not nested else "unknown"
   117→        if FN_RE.match(inner):          # a function type wrapped in parens
   118→            return conv_type(inner, ctx, nested)
   119→        if not FN_RE.match(s):
   120→            if len(top_split(inner, ",")) > 1:      # tuple in type position
   121→                return ("LuaMultiReturn<[" + ", ".join(
   122→                    conv_type(e, ctx, True)
   123→                    for e in top_split(inner, ",")) + "]>")
   124→            return conv_type(inner, ctx, True)
   125→    m = FN_RE.match(s)
   126→    if m:
   127→        params_s, ret_s = m.group(1).strip(), m.group(2).strip()
   128→        ret_s = re.sub(r"^\((.*)\)$", r"\1", ret_s).strip()
   129→        ret_ts = "void" if ret_s in ("", "()") else conv_type(ret_s, ctx, True)
   130→        pl = []
   131→        for j, pp in enumerate(top_split(params_s, ",")):
   132→            pp = pp.strip()
   133→            if not pp:
   134→                continue
   135→            if pp.startswith("..."):
   136→                rest = pp[3:].strip()
   137→                ty = rest.split(":", 1)[1].strip() if ":" in rest else "any"
   138→                pl.append((f"...args: {conv_type(ty, ctx, True)}[]", False))
   139→            elif ":" in pp:
   140→                nm, ty = pp.split(":", 1)
   141→                nm, ty = nm.strip(), ty.strip()
   142→                opt = ty.endswith("?")
   143→                ty = ty.rstrip("?").strip()
   144→                pl.append((f"{nm}: {conv_type(ty, ctx, True)}", opt))
   145→            else:
   146→                pl.append((f"_{j}: {conv_type(pp, ctx, True)}", False))
   147→        # fold mid-list optionals into their type (TS1016 applies to
   148→        # function types as well)
   149→        names = []
   150→        for i, (frag, opt) in enumerate(pl):
   151→            if opt and any(o for _, o in pl[i + 1:]):
   152→                nm, ty = frag.split(":", 1)
   153→                if not ty.strip().endswith("| undefined"):
   154→                    frag = f"{nm}: {ty.strip()} | undefined"
   155→            names.append(frag)
   156→        head = "(this: void, " + ", ".join(names) + ")" if names else "(this: void)"
   157→        return f"{head} => {ret_ts}"
   158→    if s.startswith("{") and s.endswith("}"):
   159→        inner = s[1:-1].strip()
   160→        if not inner:
   161→            return "object"
   162→        elems = top_split(inner, ",")
   163→        mm = re.match(r"^\[\s*(\w+)\s*\]\s*:\s*(.+)$", elems[0].strip(), re.S)
   164→        if len(elems) == 1 and mm and mm.group(1) not in ("string", "number"):
   165→            # a literal/union key (e.g. [Face]): TS needs a mapped type
   166→            return f"{{ [K in {mm.group(1)}]: {conv_type(mm.group(2), ctx, True)} }}"
   167→        if len(elems) == 1 and mm:
   168→            key = {"any": "string"}.get(mm.group(1), mm.group(1))
   169→            return f"{{ [key: {key}]: {conv_type(mm.group(2), ctx, True)} }}"
   170→        if len(elems) == 1 and len(top_split(elems[0], ":")) == 1 \
   171→                and not elems[0].strip().startswith("["):
   172→            return f"({conv_type(elems[0], ctx, True)})[]" if \
   173→                elems[0].strip().startswith("{") else \
   174→                f"{conv_type(elems[0], ctx, True)}[]"     # array {T} / { {T} }
   175→        fields = []
   176→        for fe in elems:
   177→            fe = fe.strip().rstrip(",").strip()
   178→            if not fe:
   179→                continue
   180→            if fe.startswith("["):
   181→                mm2 = re.match(r"^\[\s*(\w+)\s*\]\s*:\s*(.+)$", fe)
   182→                if mm2:
   183→                    key = {"any": "string"}.get(mm2.group(1), mm2.group(1))
   184→                    fields.append(f"[key: {key}]: {conv_type(mm2.group(2), ctx, True)}")
   185→                    continue
   186→            fparts = top_split(fe, ":")
   187→            if len(fparts) > 1:
   188→                nm, ty = fparts[0].strip(), ":".join(fparts[1:]).strip()
   189→                opt = nm.endswith("?")
   190→                nm = nm.rstrip("?").strip()
   191→                if ty.endswith("?"):
   192→                    opt = True
   193→                    ty = ty[:-1].strip()
   194→                tag = re.match(r"^(read|write|export|type)\s+(.+)$", nm)
   195→                if tag:
   196→                    nm = tag.group(2)
   197→                    NOTES.append((ctx.where, f"stripped '{tag.group(1)}' tag on {nm}"))
   198→                if nm in TS_KEYWORD_MEMBERS:
   199→                    nm = f'"{nm}"'
   200→                fields.append(f"{nm}{'?' if opt else ''}: {conv_type(ty, ctx, True)}")
   201→            else:
   202→                fields.append(f'"{fe}": unknown')
   203→                WARN.append((ctx.where, f"bare table field {fe!r} -> unknown"))
   204→        return "{ " + "; ".join(fields) + " }"
   205→    if s.startswith('"') and s.endswith('"'):
   206→        return s
   207→    if s in ("true", "false"):
   208→        return s
   209→    if s in PRIM_MAP:
   210→        return PRIM_MAP[s]
   211→    if s == "any":
   212→        return "any"
   213→    if s in PRIMITIVES or s in ("object",):
   214→        return s
   215→    if s in ctx.declared:
   216→        return s
   217→    WARN.append((ctx.where, f"unresolved identifier {s!r} -> unknown"))
   218→    return "unknown"

   224→def parse_signature(sig):
   225→    sig = (sig or "").strip()
   226→    rets = ""
   227→    if SIG_ARROW in sig:
   228→        sig, rets = sig.split(SIG_ARROW, 1)
   229→        sig, rets = sig.strip(), rets.strip()
   230→    m = re.match(r"^([\w.]+)([:.])([\w.]+)\s*\((.*)\)\s*$", sig, re.S)
   231→    if not m:
   232→        return None
   233→    return m.group(1), m.group(2) == ":", m.group(3), m.group(4), rets

   236→def doc_line(mem, indent="    "):
   237→    lead = (mem.get("lead") or "").strip()
   238→    if lead:
   239→        first = lead.split(". ")[0].replace("*/", "*∕").strip()
   240→        if not first.endswith("."):
   241→            first += "."
   242→        return f"{indent}/** {first} */\n"
   243→    return ""

   246→def params_to_ts(params_s, ctx):
   247→    """Signature parameter list -> [(tsName, tsType, isOptionalModifier)]."""
   248→    out = []
   249→    for j, pp in enumerate(top_split(params_s, ",")):
   250→        pp = pp.strip()
   251→        if not pp:
   252→            continue
   253→        if pp.startswith("("):
   254→            # a parenthesised parameter: a callback, a union with a callback,
   255→            # or a wrapper form — convert the whole and name it positionally
   256→            out.append((f"arg{j}", conv_type(pp, ctx, True), False))
   257→        elif pp.startswith("..."):
   258→            rest = pp[3:].strip()
   259→            ty = rest.split(":", 1)[1].strip() if ":" in rest else "any"
   260→            out.append(("...args", f"{conv_type(ty, ctx, True)}[]", False))
   261→        elif ":" in pp:
   262→            nm, ty = pp.split(":", 1)
   263→            nm, ty = nm.strip(), ty.strip()
   264→            opt = ty.endswith("?")
   265→            ty = ty.rstrip("?").strip()
   266→            out.append((nm, conv_type(ty, ctx, True), opt))
   267→        else:
   268→            out.append((f"_{j}", conv_type(pp, ctx, True), False))
   269→    # a required parameter cannot follow an optional one (TS1016): fold
   270→    # mid-list optionals into their type instead
   271→    for i, (nm, ty, opt) in enumerate(out):
   272→        if opt and any(not o2 for _, _, o2 in out[i + 1:]):
   273→            if not ty.endswith("| undefined"):
   274→                ty = f"{ty} | undefined"
   275→            out[i] = (nm, ty, False)
   276→    return [(f"{nm}{'?' if opt else ''}", ty) for nm, ty, opt in out]

   279→def returns_to_ts(rets_s, ctx):
   280→    if not rets_s:
   281→        return "void"
   282→    inner = re.sub(r"^\((.*)\)$", r"\1", rets_s.strip()).strip()
   283→    elems = top_split(inner, ",")
   284→    if len(elems) > 1:
   285→        return ("LuaMultiReturn<[" + ", ".join(conv_type(e, ctx, True)
   286→                                               for e in elems) + "]>")
   287→    return conv_type(rets_s, ctx)

   290→def balanced(s):
   291→    """True if parens/braces/brackets nest cleanly and end at depth 0."""
   292→    depth, i, instr = 0, 0, False
   293→    while i < len(s):
   294→        c = s[i]
   295→        if instr:
   296→            if c == '"':
   297→                instr = False
   298→        elif c == '"':
   299→            instr = True
   300→        elif c in "([{":
   301→            depth += 1
   302→        elif c in ")]}":
   303→            depth -= 1
   304→            if depth < 0:
   305→                return False
   306→        i += 1
   307→    return depth == 0
   308→
   309→
   310→def unwrap_balanced(s):
   311→    """Strip one outer paren pair if what remains stays balanced."""
   312→    s = s.strip()
   313→    if not (s.startswith("(") and s.endswith(")")):
   314→        return s
   315→    depth, i, instr = 0, 0, False
   316→    while i < len(s):
   317→        c = s[i]
   318→        if instr:
   319→            if c == '"':
   320→                instr = False
   321→        elif c == '"':
   322→            instr = True
   323→        elif c in "([{":
   324→            depth += 1
   325→        elif c in ")]}":
   326→            depth -= 1
   327→            if depth == 0 and i < len(s) - 1:
   328→                return s            # closes before the end: not a wrapper
   329→        i += 1
   330→    return s[1:-1].strip() if depth == 0 else s

   333→def member_decl(parent, name, mem, ctx):
   334→    kind = mem.get("kind")
   335→    sig = (mem.get("signature") or "").strip()
   336→    ctx.where = f"{parent}.{name}"
   337→    if kind == "field":
   338→        m = re.match(r"^[\w.]+\.[\w.]+\s*:\s*(.+)$", sig)
   339→        if not m:
   340→            WARN.append((ctx.where, f"field type unparsed: {sig!r}"))
   341→            nm = f'"{name}"' if name in TS_KEYWORD_MEMBERS else name
   342→            return f"{doc_line(mem)}{nm}: unknown"
   343→        raw = m.group(1).strip()
   344→        # a documented optional field ("field: T ?") -> optional property
   345→        opt = raw.endswith("?")
   346→        if opt:
   347→            raw = raw[:-1].strip()
   348→        ty = conv_type(raw, ctx)
   349→        nm = f'"{name}"' if name in TS_KEYWORD_MEMBERS else name
   350→        if opt and ty not in ("unknown", "any"):
   351→            return f"{doc_line(mem)}{nm}?: {ty}"
   352→        return f"{doc_line(mem)}{nm}: {ty}"
   353→    nm = f'"{name}"' if name in TS_KEYWORD_MEMBERS else name
   354→    parsed = parse_signature(sig)
   355→    if parsed is None:
   356→        WARN.append((ctx.where, f"UNPARSED signature {sig!r}"))
   357→        return f"// UNPARSED {ctx.where}: {sig}"
   358→    recv, colon, name_p, params_s, rets_s = parsed
   359→
   360→    # Overloaded signatures: the crawl renders them as
   361→    #   receiver.name(<form A>) -> & (<form B>)
   362→    # — every form is a full function type. Emit one overload per form.
   363→    if "&" in rets_s and len(top_split(rets_s, "&")) > 1:
   364→        forms = [params_s] + [p for p in top_split(rets_s, "&") if p.strip()]
   365→        lines = []
   366→        for form in forms:
   367→            fm = None
   368→            for cand in (unwrap_balanced(form.strip()), form.strip()):
   369→                m = FN_RE.match(cand)
   370→                if m and balanced(m.group(1)):
   371→                    fm = m
   372→                    break
   373→            if not fm:
   374→                WARN.append((ctx.where, f"overload form unparsed: {form!r}"))
   375→                continue
   376→            pl = params_to_ts(fm.group(1), ctx)
   377→            ret_ts = returns_to_ts(fm.group(2), ctx)
   378→            args = ", ".join(f"{k}: {v}" for k, v in pl)
   379→            if kind == "method":
   380→                lines.append(f"{doc_line(mem)}{nm}({args}): {ret_ts}")
   381→            else:
   382→                self_dot = "this: void" + (", " if args else "")
   383→                lines.append(f"{doc_line(mem)}{nm}({self_dot}{args}): {ret_ts}")
   384→        return "\n".join(lines)
   385→
   386→    ret_ts = returns_to_ts(rets_s, ctx)
   387→
   388→    if parent in SPEC_TYPES and kind == "function" \
   389→            and FN_RE.match(params_s.strip()):
   390→        # spec handler: a user-assigned callback field; params_s IS its type
   391→        cb_nm = f'"{name}"' if name in TS_KEYWORD_MEMBERS else name
   392→        return f"{doc_line(mem)}{cb_nm}?: {conv_type(params_s, ctx)}"
   393→
   394→    pl = params_to_ts(params_s, ctx)
   395→    args = ", ".join(f"{k}: {v}" for k, v in pl)
   396→    if kind == "method":
   397→        return f"{doc_line(mem)}{nm}({args}): {ret_ts}"
   398→    self_dot = "this: void" + (", " if args else "")
   399→    return f"{doc_line(mem)}{nm}({self_dot}{args}): {ret_ts}"

   420→def emit_interface(name, members, doc=""):
   421→    if name in interfaces or name in aliases:
   422→        raise SystemExit(f"collision: {name}")
   423→    interfaces[name] = members
   424→    if doc:
   425→        docs[name] = doc
   426→
   427→
   428→# ------------------------------------------- pass 1: groups -> interfaces
   429→for gname, mems in M.items():
   430→    if gname in ("vec3", "color", "Faces"):
   431→        continue                # constructors / Faces are emitted specially
   432→    iname = GLOBAL_GROUP.get(gname, gname)
   433→    lines = [member_decl(gname, mn, mems[mn], ctx) for mn in sorted(mems)]
   434→    emit_interface(iname, lines, (P.get(gname, {}).get("lead") or "").strip())
   435→    if gname in GLOBAL_GROUP:
   436→        emitted_globals.append((gname, iname))
   437→
   438→STATS["member groups -> interfaces"] = len(M) - 3
   439→STATS["API members converted"] = sum(len(v) for v in M.values())
   440→
   441→# ------------------------------------------- pass 2: typedef-only parents
   442→OPAQUE = []
   443→for k in sorted(P):
   444→    if k in M or k in ("vec3", "color", "message", "tween", "camera", "task",
   445→                      "Vec3", "Color", "TimeName", "Faces", "DataView"):
   446→        continue
   447→    # skip lowercase twin stubs of PascalCase groups (world/World twins
   448→    # whose canonical page carries the members)
   449→    if k[0].islower() and k.capitalize() in M:
   450→        continue
   451→    td = (P[k].get("typedef") or "").strip()
   452→    lead = (P[k].get("lead") or "").strip()
   453→    ctx.where = f"typedef:{k}"
   454→    if not td:
   455→        emit_interface(k, [], f"{lead} [no documented members: opaque]" if lead
   456→                      else "[no documented members: opaque]")
   457→        OPAQUE.append(k)
   458→        continue
   459→    before = len(WARN)
   460→    ts = conv_type(td, ctx, True)
   461→    new = WARN[before:]
   462→    garbled = td.endswith("|") or td.startswith("|") or \
   463→        any("unresolved identifier" in w[1] for w in new) or not ts
   464→    if garbled:
   465→        emit_interface(k, [], f"{lead} [typedef not machine-convertible in "
   466→                      f"the source extract: opaque]" if lead
   467→                      else "[typedef not machine-convertible: opaque]")
   468→        OPAQUE.append(k)
   469→    else:
   470→        aliases[k] = ts
   471→
   472→STATS["typedef aliases"] = len(aliases)
   473→STATS["opaque interfaces"] = len(OPAQUE)
   474→
   475→# ------------------------------------------------------------------ assembly
   476→def render_interface(name, members, doc=""):
   477→    out = []
   478→    if doc:
   479→        out.append(f"/** {doc.replace('*/', '*∕')} */")
   480→    out.append(f"interface {name} {{")
   481→    for m in members:
   482→        for ln in m.split("\n"):
   483→            out.append(f"    {ln}" if ln.strip() else "")
   484→    out.append("}")
   485→    return "\n".join(out)

   488→VERIFIED = '''// ===========================================================================
   489→// VERIFIED SUPPLEMENT — members the reference section does not carry, taken
