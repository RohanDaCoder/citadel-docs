   144→FM_RE = re.compile(r"^---\n(.*?)\n---\n", re.S)

   147→def parse_page(path):
   148→    raw = open(path).read()
   149→    m = FM_RE.match(raw)
   150→    if not m:
   151→        raise SystemExit(f"no frontmatter: {path}")
   152→    fm = {}
   153→    for line in m.group(1).split("\n"):
   154→        if ":" in line:
   155→            k, v = line.split(":", 1)
   156→            v = v.strip()
   157→            if v.startswith("[") and v.endswith("]"):
   158→                v = [x.strip().strip('"') for x in v[1:-1].split(",") if x.strip()]
   159→            fm[k.strip()] = v
   160→    body = raw[m.end():].lstrip("\n")
   161→    return fm, body

   164→def yaml_str(value):
   165→    """Single-quoted YAML scalar."""
   166→    return "'" + str(value).replace("'", "''").replace("\n", " ").strip() + "'"

   169→def frontmatter(pairs):
   170→    lines = ["---"]

   241→def original_url_to_route(href):
   242→    """Map an original minefort docs URL to a new Starlight route."""
   243→    if not href or not href.startswith("/docs/citadel"):
   244→        return None
   245→    rest = href[len("/docs/citadel"):].strip("/")
   246→    if not rest:
   247→        return "/"
   248→    parts = rest.split("/")
   249→    if parts[0] == "guides" and len(parts) > 1:
   250→        g = parts[1]
   251→        return wiki_route("quick-start" if g == "your-first-pack" else f"guide-{g}")
   252→    if parts[0] == "reference":
   253→        if len(parts) == 1:
   254→            return "/ref/"
   255→        if len(parts) == 2:
   256→            return ref_route(parts[1])
   257→        return ref_route(parts[1], parts[2])
   258→    if parts[0] in DOC_TO_WIKI:
   259→        w = DOC_TO_WIKI[parts[0]]
   260→        return "/ref/" if w == "" else wiki_route(w)
   261→    return None

   272→def rewrite_links(body, ctx):
   273→    def wiki_sub(m):
   274→        slug, anchor = m.group(2), m.group(3) or ""
   275→        route = wiki_route(slug)
   276→        if route is None:
   277→            WARNINGS.append(f"{ctx}: /wiki/{slug} has no target page")
   278→            return m.group(0)
   279→        return f"]({route}{anchor})"
   280→
   281→    def app_sub(m):
   282→        slug, anchor = m.group(2), m.group(3) or ""
   283→        route = wiki_route(slug) if SLUG2SECTION.get(slug) == "appendix" else None
   284→        if route is None:
   285→            WARNINGS.append(f"{ctx}: /appendix/{slug} has no target page")
   286→            return m.group(0)
   287→        return f"]({route}{anchor})"
   288→
   289→    def ref_sub(m):
   290→        parent, member, anchor = m.group(2), m.group(3), m.group(4) or ""
   291→        route = ref_route(parent, member)
   292→        if route is None:
   293→            WARNINGS.append(f"{ctx}: /ref/{parent}/{member or ''} has no target page")
   294→            return m.group(0)
   295→        return f"]({route}{anchor})"
   296→
   297→    body = WIKI_LINK_RE.sub(wiki_sub, body)
   298→    body = APP_LINK_RE.sub(app_sub, body)
   299→    body = REF_LINK_RE.sub(ref_sub, body)
   300→    body = body.replace("](/ref)", "](/ref/)")
   301→    return body

   304→CALLOUT_TYPES = {"In short": "note", "Note": "note", "Tip": "tip",
   305→                 "Gotcha": "caution", "Warning": "danger"}
   306→CALLOUT_RE = re.compile(r"^> \*\*(In short|Note|Gotcha|Tip|Warning):\*\*(.*)$")
   307→
   308→
   309→def convert_callouts(body):
   310→    """Convert `> **Type:** ...` blockquotes to Starlight ::: admonitions."""
   311→    lines = body.split("\n")
   312→    out = []
   313→    i = 0
   314→    while i < len(lines):
   315→        m = CALLOUT_RE.match(lines[i])
   316→        if not m:
   317→            out.append(lines[i])
   318→            i += 1
   319→            continue
   320→        ctype = CALLOUT_TYPES[m.group(1)]
   321→        title = m.group(1) if m.group(1) in ("In short", "Gotcha") else None
   322→        opener = f":::{ctype}" + (f"[{title}]" if title else "")
   323→        block = [m.group(2).strip()]
   324→        i += 1
   325→        while i < len(lines) and lines[i].startswith(">"):
   326→            block.append(re.sub(r"^> ?", "", lines[i]))
   327→            i += 1
   328→        # trim trailing blank lines inside the block
   329→        while block and not block[-1].strip():
   330→            block.pop()
   331→        out.append(opener)
   332→        out.extend(block)
   333→        out.append(":::")
   334→        if i < len(lines) and lines[i].strip():
   335→            out.append("")
   336→    return "\n".join(out)

   338→
   339→def strip_h1(body, title):
   340→    m = re.match(r"^# (.+)$", body, re.M)
   341→    if m and m.group(1).strip() == title.strip():
   342→        body = body[m.end():].lstrip("\n")
   343→    return body

   346→def convert_body(body, title, ctx):
   347→    body = strip_h1(body, title)
   348→    body = rewrite_links(body, ctx)
   349→    body = convert_callouts(body)
   350→    return body.strip() + "\n"

   403→def gen_event_catalog():
   404→    rows = []
   405→    seen = set()
   406→    for parent, mems in MEMBERS.items():
   407→        for name, m in mems.items():
   408→            if not name.startswith("on_"):
   409→                continue
   410→            key = f"{parent}.{name}"
   411→            if key in seen:
   412→                continue
   413→            seen.add(key)
   414→            lead = (m.get("lead", "") or "").strip()
   415→            if not lead:
   416→                p = PARENTS.get(parent)
   417→                if p:
   418→                    for mm in p.get("members", []):
   419→                        if mm["name"] == name:
   420→                            lead = mm.get("description", "")
   421→                            break
   422→            rows.append({"parent": canon(parent), "name": name,
   423→                         "signature": m.get("signature", ""), "description": lead})
   424→    rows.sort(key=lambda r: (r["parent"].lower(), r["name"]))
   425→    lines = [
   426→        "# Event catalog",
   427→        "",
   428→        "> **In short:** every `on_*` subscription in the API, one row each.",
   429→        "> Subscribe with the call, keep the connection, cancel to stop. The",
   430→        "> model itself — deadlines, asks, ordering — is in",
   431→        "> [The event model](/wiki/events).",
   432→        "",
   433→        f"{len(rows)} events across {len(set(r['parent'] for r in rows))} types.",
   434→        "",
   435→    ]
   436→    byp = OrderedDict()
   437→    for r in rows:
   438→        byp.setdefault(r["parent"], []).append(r)
   439→    for parent in sorted(byp, key=str.lower):
   440→        lines.append(f"## `{parent}`")
   441→        lines.append("")
   442→        lines.append("| Event | Signature | Fires when |")
   443→        lines.append("| --- | --- | --- |")
   444→        for r in byp[parent]:
   445→            desc = esc_cell(r["description"])
   446→            if len(desc) > 160:
   447→                desc = desc[:157] + "..."
   448→            sig = esc_cell(r["signature"])
   449→            lines.append(
   450→                f"| [`{r['name']}`](/ref/{parent}/{r['name']}) | `{sig}` | {desc} |")
   451→        lines.append("")
   452→    lines += [
   453→        "## See also",
   454→        "",
   455→        "- [The event model](/wiki/events) — subscriptions, asks, deadlines.",
   456→        "- [Runtime](/wiki/runtime) — the event queue and its 4096 cap.",
   457→        "",
   458→    ]
   459→    return ("\n".join(lines),
   460→            "Every on_* event in the Citadel API in one table — what fires it, "
   461→            "what the handler receives, and where the full treatment lives.")

   826→def page_headings(path):
   827→    heads = set()
   828→    in_code = False
   829→    for line in open(path):
   830→        if line.startswith("```"):
   831→            in_code = not in_code
   832→            continue
   833→        if in_code:
   834→            continue
   835→        m = re.match(r"^(#{1,6})\s+(.*)$", line)
   836→        if m:
   837→            heads.add(github_slug(m.group(2)))
   838→    return heads
